* Overview

This is a portable, narrative-driven, and maintainable literate configuration for Emacs.
It uses [[https://github.com/progfolio/elpaca][Elpaca]] as the package manager, `evil` for modal editing, and provides Python andat/ChatGPT integration, plus modern completions and a beautiful, usable UI.
With a single `config.org` and a local Makefile, you can bootstrap, update, or clean your setup *anywhere*.

#+begin_src emacs-lisp
  ;; custom dirs
  (defun make-var-dir-pair (variable dir &optional path)
    ;; let*, not let: TARGET must see the PATH bound on the line above. With a plain
    ;; let it saw the original nil PATH, so .data/.config resolved against the startup
    ;; directory (~) instead of the config directory, for years.
    (let* ((path (or path user-emacs-directory))
           (target (expand-file-name dir path)))
      (unless (file-directory-p target) (make-directory target))
      (set variable target)))
  (defun make-var-dir-pairs-from-list (list)
    (dolist (pair list) (apply 'make-var-dir-pair pair)))
  (make-var-dir-pairs-from-list
   '((user-local-tmp-config-dir ".config") 
     (user-local-tmp-data-dir ".data"))) 
  (when (fboundp 'startup-redirect-eln-cache)
    (startup-redirect-eln-cache
     (convert-standard-filename
      (expand-file-name "eln-cache" user-local-tmp-data-dir))))
  (setq package-user-dir (file-name-concat user-local-tmp-config-dir "elpa"))

** shallow cloning
Elpaca 0.12 removed =elpaca-git-clone-arguments=. Clone depth is now a recipe default.
0.12's own default is =:depth treeless= (a partial clone that fetches trees lazily at
checkout), which proved fragile against GitHub HTTP/2 stream resets during a cold
bootstrap: a failed checkout leaves a hollow repo that Elpaca then fails to re-clone.
Plain shallow clones fetch one pack and either succeed or leave nothing behind.
#+begin_src emacs-lisp
  (setq elpaca-order-defaults (list :type 'git :protocol 'https :inherit t :depth 1))
  ;; NB: do not set `elpaca-queue-limit'. In 0.12 (Oct 2026) throttling corrupted
  ;; dependency status tracking: finished packages were reported as failed.
#+end_src

** Elpaca package management
Here we set up Elpaca so that all packages and builds are stored *locally* in the repo directory. 
If you clone this somewhere else, everything just works.

#+begin_src emacs-lisp 
  ;;; -*- lexical-binding: t; -*-

  ;; !!!!! this file is genrated from config.org -- edit it with the understading that it will likely get overwritten !!!! 
  ;; elpaca
  (defvar elpaca-installer-version 0.12)
  (defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
  (defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
  (defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
  (defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                                :ref nil :depth 1 :inherit ignore
                                :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                                :build (:not elpaca-activate)))
  (let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
         (build (expand-file-name "elpaca/" elpaca-builds-directory))
         (order (cdr elpaca-order))
         (default-directory repo))
    (add-to-list 'load-path (if (file-exists-p build) build repo))
    (unless (file-exists-p repo)
      (make-directory repo t)
      (when (<= emacs-major-version 28) (require 'subr-x))
      (condition-case-unless-debug err
          (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                    ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                    ,@(when-let* ((depth (plist-get order :depth)))
                                                        (list (format "--depth=%d" depth) "--no-single-branch"))
                                                    ,(plist-get order :repo) ,repo))))
                    ((zerop (call-process "git" nil buffer t "checkout"
                                          (or (plist-get order :ref) "--"))))
                    (emacs (concat invocation-directory invocation-name))
                    ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                          "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                    ((require 'elpaca))
                    ((elpaca-generate-autoloads "elpaca" repo)))
              (progn (message "%s" (buffer-string)) (kill-buffer buffer))
            (error "%s" (with-current-buffer buffer (buffer-string))))
        ((error) (warn "%s" err) (delete-directory repo 'recursive))))
    (unless (require 'elpaca-autoloads nil t)
      (require 'elpaca)
      (elpaca-generate-autoloads "elpaca" repo)
      (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
  (add-hook 'after-init-hook #'elpaca-process-queues)
  (elpaca `(,@elpaca-order))
  (elpaca elpaca-use-package (elpaca-use-package-mode))
  (setq use-package-always-ensure t)
  (elpaca-wait)
#+end_src

    "Like `use-package' but accounting for asynchronous installation.

    (declare (indent defun))
    `(use-package ,name
       :ensure nil

** Terminal: ghostel
Replaces vterm + multi-vterm (2026-10). ghostel is a terminal built on libghostty-vt
(Ghostty's VT engine). Same approach as vterm but: no local toolchain (it downloads a
prebuilt module on first use, =M-x ghostel-download-module= to redo), faster, Kitty
keyboard protocol, OSC 8 links, synchronized output (helps Claude Code redraws),
auto shell integration for zsh. Prefix arg to =ghostel= opens another terminal;
=consult-ghostel= picks between them. Known: ESC goes to alt-screen apps (vim, Claude
Code fullscreen) by default; =C-c C-r= toggles that, =C-c ESC= forces normal state.
#+begin_src emacs-lisp
  (use-package ghostel
    :commands (ghostel ghostel-project ghostel-other)
    :custom
    ;; keep the downloaded module out of Elpaca's build tree (rebuilds would delete it)
    (ghostel-module-directory (expand-file-name "ghostel-module" user-local-tmp-data-dir))
    (ghostel-buffer-name-function #'ghostel-buffer-name-by-directory)
    ;; if a TUI misbehaves under TERM=xterm-ghostty, try: (ghostel-term "xterm-256color")
    )

  (use-package evil-ghostel
    :after (ghostel evil)
    :hook (ghostel-mode . evil-ghostel-mode))

  ;; No :after / after-init hook here: Elpaca evaluates this form after init has
  ;; already run, so the upstream `after-init' hook never fires. The binding is an
  ;; autoload; the mode (consult-buffer sources, bookmarks) is enabled on first load.
  (use-package consult-ghostel

           :map ghostel-semi-char-mode-map

** some themes
I switch themes often (=consult-theme=), so org cosmetics that depend on the palette are
derived from the current background and re-applied by =enable-theme-functions= and
=disable-theme-functions= whenever a theme changes, rather than hard-coded for one theme.
The theme packages below are installed but none is loaded at startup; that costs only a
load-path entry each, nothing is evaluated until a theme is chosen.
- ghost stars: leading heading stars in a faint tint (leuven-style) instead of invisible,
  so the depth is still readable behind the org-modern glyph. 18 = percent shift.
- source blocks: a subtle background so they stand out. 4 = percent shift.
- headings: inherit =variable-pitch= (ETBookOT), normal weight, and a fixed size ladder
  (1.6 title, then 1.4 / 1.25 / 1.15 / 1.05, 1.0 below) under every theme; colours stay the theme's.
#+begin_src emacs-lisp
  (defun my/org-face-tweaks (&rest _)
    "Ghost leading stars and a subtle source-block background, for any theme.
Org's faces only exist once org is loaded, so faces that are not defined yet
are skipped; the `org' eval-after-load below re-runs this when they appear."
    (require 'color)
    (when-let* ((bg (face-background 'default nil t))
                (rgb (color-name-to-rgb bg)) ; nil on tty/daemon frames ("unspecified-bg")
                (shift (if (color-dark-p rgb) #'color-lighten-name #'color-darken-name)))
      (when (facep 'org-hide)
        (set-face-attribute 'org-hide nil :foreground (funcall shift bg 18)))
      (when (facep 'org-block)
        (set-face-attribute 'org-block nil :background (funcall shift bg 4)))
      ;; Headings under every theme: variable-pitch (ETBookOT, set in early-init),
      ;; never bold, and my own size ladder. The theme keeps the colours only.
      ;; `variable-pitch' is appended to each face's inherit list, not substituted.
      (dolist (spec '((org-document-title . 1.6)
                      (org-level-1 . 1.4) (org-level-2 . 1.25) (org-level-3 . 1.15)
                      (org-level-4 . 1.05) (org-level-5 . 1.0) (org-level-6 . 1.0)
                      (org-level-7 . 1.0) (org-level-8 . 1.0)))
        (let ((f (car spec)))
          (when (facep f)
            (let ((inh (face-attribute f :inherit)))
              (set-face-attribute
               f nil
               :weight 'normal
               :height (cdr spec)
               :inherit (delete-dups (append (cond ((memq inh '(unspecified nil)) nil)
                                                   ((listp inh) inh)
                                                   (t (list inh)))
                                             '(variable-pitch))))))))))
  (add-hook 'enable-theme-functions #'my/org-face-tweaks)
  ;; Disabling a theme recalculates only the faces THAT theme defined, so after
  ;; e.g. monokai -> "default" org-hide/org-block fall back to stock specs while
  ;; after tango -> "default" they keep our values. Re-derive on disable as well.
  (add-hook 'disable-theme-functions #'my/org-face-tweaks)
  (with-eval-after-load 'org
    (my/org-face-tweaks)
    ;; `org-mode' resets `org-hide' to the frame background on EVERY activation
    ;; (org.el: "Try to set `org-hide' face correctly"), which would undo the ghost
    ;; stars for each newly opened org file. Re-apply after org has had its say.
    (add-hook 'org-mode-hook #'my/org-face-tweaks)
    ;; Ghost-shade the #+ prefix of keywords and block delimiters with the same
    ;; `org-hide' tint as the leading stars (org-modern can only hide or replace it).
    (font-lock-add-keywords 'org-mode '(("^[ \t]*\\(#\\+\\)" 1 'org-hide prepend)) 'append))
#+end_src

** Auto-Tangle and Live Reload

Every time you save `config.org`, it re-tangles and reloads itself. 
This means your Emacs stays in sync with your org file, making true literate configuration seamless.

#+begin_src emacs-lisp 
  (defun org-babel-tangle-config ()
    (interactive)
    (when (string-equal (buffer-file-name)
                        (expand-file-name "config.org" user-emacs-directory))
      (org-babel-tangle)
      (load-file (expand-file-name "init.el" user-emacs-directory))
      (elpaca-process-queues)))

  (add-hook 'org-mode-hook
            (lambda ()
              (when (equal (buffer-file-name) ; only add hook if it's config.org
                           (expand-file-name "config.org" user-emacs-directory))
                (add-hook 'after-save-hook #'org-babel-tangle-config nil t))))

  (setq org-confirm-babel-evaluate nil)
#+end_src

* Makefile for Bootstrapping and Cleanup
:PROPERTIES:
:header-args: :tangle Makefile
:END:

Run `make help` for usage.

#+begin_src makefile 
  EMACS ?= emacs
  CONFIG ?= config.org
  CURDIR := $(shell pwd)

  .PHONY: all bootstrap clean-packages clean-all clean help 

  all: bootstrap

  bootstrap:
  	@echo "Tangling org configuration..."
  	$(EMACS) --batch -l org --eval '(org-babel-tangle-file "$(CONFIG)")'
  	@echo "Bootstrap complete."

  clean-packages:
  	rm -rf ./elpaca
  	@echo "All packages in ./elpaca/ cleared."

  clean-all:
  	@echo "Removing all files/dirs except .git, .gitignore, config.org, Makefile..."
  	@find . -mindepth 1 -maxdepth 1 \
  	  ! -name '.git' \
  	  ! -name '.gitignore' \
  	  ! -name 'config.org' \
  	  ! -name 'Makefile' \
  	  -exec rm -rf {} +
  	@echo "Clean-all complete."

  clean: clean-packages
  	rm -f ./init.el ./early-init.el ./lisp/*.el
  	@echo "Configuration cleaned. Only config.org and Makefile remain."

  help:
  	@echo "Emacs Configuration Management:"
  	@echo ""
  	@echo "  make bootstrap        Tangle and set up the Emacs configuration from config.org."
  	@echo "  make clean-packages   Remove all installed packages (forces reinstallation on next bootstrap)."
  	@echo "  make clean            Remove all generated configuration files, leaving only config.org and Makefile."
  	@echo "  make help             Show this help message."
#+end_src
